cmake_minimum_required(VERSION 3.24)
project(veyra_radeon_backend_tests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "These tests require native Windows and D3D12 (no WSL).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(VEYRA_ROOT "../.." ABSOLUTE)
set(VEYRA_AMF_SDK_ROOT "" CACHE PATH "AMF SDK directory containing include/core and include/components")
set(VEYRA_FIDELITYFX2_ROOT "" CACHE PATH "FidelityFX SDK Kits/FidelityFX directory")
add_compile_definitions(UNICODE _UNICODE _WINDOWS WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra)
  add_link_options(-static)
endif()
include_directories("${VEYRA_ROOT}/include")
enable_testing()
add_library(radeon_test_base STATIC "${VEYRA_ROOT}/src/base/Log.cpp" "${VEYRA_ROOT}/src/base/RuntimePaths.cpp")
add_executable(radeon_nr_stack_refine_gpu NrStackRefineGpuTests.cpp)
target_link_libraries(radeon_nr_stack_refine_gpu PRIVATE radeon_test_base d3d12 dxgi)

# Independent offline NR experiment, using the exact production adapter and
# driver preflight. Runtime/model assets stay outside the source/build tree.
add_executable(radeon_nr_probe RadeonNrProbe.cpp
  "${VEYRA_ROOT}/src/pipeline/LmxxfNrBackend.cpp"
  "${VEYRA_ROOT}/src/pipeline/AmdHipPreflight.cpp"
  "${VEYRA_ROOT}/src/gfx/GpuSchedulingPriority.cpp")
target_link_libraries(radeon_nr_probe PRIVATE radeon_test_base d3d12 dxgi dxguid)
if(MINGW)
  target_link_options(radeon_nr_probe PRIVATE -municode)
endif()

# No SDK macros: the exact production stub paths must report unavailable.
add_executable(radeon_build_capabilities BuildCapabilitiesTests.cpp
  "${VEYRA_ROOT}/src/gfx/AmfFrcBackend.cpp" "${VEYRA_ROOT}/src/gfx/FsrSrBackend.cpp"
  "${VEYRA_ROOT}/src/gfx/FsrFgPresenter.cpp" "${VEYRA_ROOT}/src/guidance/AmdOpticalFlow.cpp")
target_link_libraries(radeon_build_capabilities PRIVATE radeon_test_base d3d12 dxgi dxguid)
add_test(NAME radeon_build_capabilities COMMAND radeon_build_capabilities)
add_executable(radeon_selection "${VEYRA_ROOT}/tests/unit/RadeonSelectionTests.cpp")
add_test(NAME radeon_selection COMMAND radeon_selection)
add_executable(radeon_effect_availability "${VEYRA_ROOT}/tests/unit/EffectAvailabilityTests.cpp")
add_test(NAME radeon_effect_availability COMMAND radeon_effect_availability)

if(VEYRA_AMF_SDK_ROOT)
  if(NOT EXISTS "${VEYRA_AMF_SDK_ROOT}/include/components/FRC.h")
    message(FATAL_ERROR "VEYRA_AMF_SDK_ROOT must contain include/components/FRC.h (AMF repository: amf/public)")
  endif()
  add_executable(radeon_amf_gpu AmfFrcGpuTests.cpp "${VEYRA_ROOT}/src/gfx/AmfFrcBackend.cpp")
  target_compile_definitions(radeon_amf_gpu PRIVATE VEYRA_HAS_AMF=1)
  target_include_directories(radeon_amf_gpu PRIVATE "${VEYRA_AMF_SDK_ROOT}/include")
  target_link_libraries(radeon_amf_gpu PRIVATE radeon_test_base d3d12 dxgi dxguid ole32)
  add_test(NAME radeon_amf_gpu COMMAND radeon_amf_gpu)
  set_tests_properties(radeon_amf_gpu PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(VEYRA_FIDELITYFX2_ROOT)
  if(NOT EXISTS "${VEYRA_FIDELITYFX2_ROOT}/api/include/ffx_api_loader.h")
    message(FATAL_ERROR "VEYRA_FIDELITYFX2_ROOT must point to Kits/FidelityFX")
  endif()
  add_library(radeon_fsr STATIC "${VEYRA_ROOT}/src/gfx/FsrSrBackend.cpp" "${VEYRA_ROOT}/src/gfx/FsrFgPresenter.cpp")
  target_compile_definitions(radeon_fsr PRIVATE VEYRA_HAS_FSR=1)
  target_include_directories(radeon_fsr PRIVATE "${VEYRA_FIDELITYFX2_ROOT}/api/include"
    "${VEYRA_FIDELITYFX2_ROOT}/upscalers/include" "${VEYRA_FIDELITYFX2_ROOT}/framegeneration/include")
  target_link_libraries(radeon_fsr PUBLIC radeon_test_base d3d12 dxgi dxguid)
  add_executable(radeon_fsr_lifecycle FsrSrLifecycleTests.cpp)
  target_link_libraries(radeon_fsr_lifecycle PRIVATE radeon_fsr)
  add_test(NAME radeon_fsr_lifecycle COMMAND radeon_fsr_lifecycle)
  set_tests_properties(radeon_fsr_lifecycle PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(VEYRA_AMF_SDK_ROOT AND VEYRA_FIDELITYFX2_ROOT)
  add_executable(radeon_pipeline_probe RadeonPipelineProbe.cpp "${VEYRA_ROOT}/src/gfx/AmfFrcBackend.cpp")
  target_compile_definitions(radeon_pipeline_probe PRIVATE VEYRA_HAS_AMF=1)
  target_include_directories(radeon_pipeline_probe PRIVATE "${VEYRA_AMF_SDK_ROOT}/include")
  target_link_libraries(radeon_pipeline_probe PRIVATE radeon_fsr ole32)
  # Explicit invocation keeps performance diagnostics out of routine CTest.
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES TIMEOUT 180)
